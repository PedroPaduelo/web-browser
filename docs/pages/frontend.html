⚛️ Frontend (React)

Visão Geral

Frontend em React + TypeScript com Vite. Responsável pela interface de usuário, exibição de vídeo e captura de eventos de input.

│   ├── AdminPanel.tsx    # Painel admin
│   └── LoginScreen.tsx   # Login com API Key

│   ├── BrowserViewer.tsx # Canvas + Audio

│   └── SessionControls.tsx

│   ├── useSession.ts     # CRUD de sessões via REST

└── types/
    └── index.ts

        img.src = url

    return () => ws.disconnect()
  }, [sessionId])

  return { canvasRef }
}

Use Input Relay Hook

function useInputRelay(sessionId: string) {
  const ws = useMemo(() => {
    const client = createWsClient()
    client.connect(getAuthWsUrl('/ws/input', sessionId))
    return client
  }, [sessionId])

  useEffect(() => {
    const canvas = document.getElementById('canvas')
    if (!canvas) return

    const handlePointerMove = (e: PointerEvent) => {
      const rect = canvas.getBoundingClientRect()
      const x = e.clientX - rect.left
      const y = e.clientY - rect.top
      ws.send({
        type: 'input',
        sessionId,
        data: { type: 'mouse_move', x, y }
      })
    }

    canvas.addEventListener('pointermove', handlePointerMove)
    canvas.addEventListener('pointerdown', handleClick)
    window.addEventListener('keydown', handleKeyDown)
    window.addEventListener('keyup', handleKeyUp)

    return () => {
      canvas.removeEventListener('pointermove', handlePointerMove)
      // ... outros removes
    }
  }, [sessionId, ws])

  return ws
}

Browser Viewer

Componente principal que combina Canvas (vídeo) + Audio API + Input relay:

export default function BrowserViewer({ sessionId }) {
  const { canvasRef } = useStream(sessionId)
  useAudio(sessionId)
  useInputRelay(sessionId)

  return (
    <div className="viewer-container">
      <canvas ref={canvasRef} width={1280} height={720} />
      <ConnectionStatus />
    </div>
  )
}
